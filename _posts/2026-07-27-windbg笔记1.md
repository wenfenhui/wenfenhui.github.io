## 安装windbg-mcp

我用的是下面这个,不够这个我全程叫agent帮我配置：

https://github.com/svnscha/mcp-windbg/releases/tag/v0.15.0  这是安装的前置条件

- Windows 系统 ，并安装 Debugging Tools for Windows 或从 Microsoft Store 安装 WinDbg（需要其中的 cdb.exe ）
- Python 3.10+
- 任意 MCP 兼容客户端：GitHub Copilot、Claude Desktop、Cline、Cursor、Windsurf 等

mcp-windbg 本质上是一个 Python 包装器，通过 stdin/stdout 与 CDB 进程交互，执行命令并返回结果。

**重点来了，如果我们需要让windbg gui能感应到他能调试，需要让 WinDbg 作为 调试服务器 ，然后 mcp-windbg 通过远程协议连上去**

我们在命令行当中输入：.server tcp:port=5005    现在下面就成功了，然后问agent就可以调试了~

#include <stdio.h>
#include <tchar.h>

int _tmain(int argc, _TCHAR* argv[]) {
    const char* p = "hello world";
    printf("%s\n", p);

    // 调试断点
#ifdef _M_IX86
    __asm int 3;
#else
    __debugbreak();
#endif

    return 0;
}

涉及到结构体dt的章节

```
#include <stdio.h>
#include <tchar.h>
#include <string>

// ============================================================
// WinDbg d 命令系列演示项目
// 可用命令: dt / db / dw / dd / dq / da / du / dc / dS / dW
//          dy / dpp / dpu / dpa / dps / ds / d*s
// 建议编译配置: Debug | x64 (便于查看完整符号信息)
// ============================================================

// ---------- 1. 枚举 (enum) ----------
// dt WinDbgDemo!Color
enum Color {
    RED = 0,
    GREEN = 1,
    BLUE = 2,
    YELLOW = 3
};

// ---------- 2. 简单结构体 ----------
// dt WinDbgDemo!Point
struct Point {
    int x;
    int y;
};

// ---------- 3. 带数组的结构体 ----------
// dt WinDbgDemo!Buffer
struct Buffer {
    char data[16];
    int  length;
    int  capacity;
};

// ---------- 4. 嵌套结构体 ----------
// dt WinDbgDemo!Rectangle
struct Rectangle {
    Point topLeft;       // 嵌套结构体
    Point bottomRight;
    Color color;         // 枚举成员
};

// ---------- 5. 链表节点 (双向链表) ----------
// dt WinDbgDemo!Node
struct Node {
    int   value;
    Node* next;          // 指向下一个节点 (dpp/dpu 可解引用)
    Node* prev;          // 指向前一个节点
};

// ---------- 6. 带位域的结构体 ----------
// dt WinDbgDemo!Flags
struct Flags {
    unsigned int visible : 1;
    unsigned int enabled : 1;
    unsigned int paused : 1;
    unsigned int reserved : 5;
    unsigned int priority : 8;
};

// ---------- 7. 联合体 (union) ----------
// dt WinDbgDemo!Data
union Data {
    int   i;
    float f;
    char  c[4];
};

// ---------- 8. 带虚函数的类 (观察 vtable) ----------
// dt WinDbgDemo!Animal
class Animal {
public:
    Animal(int age, const char* name) : age_(age), name_(name) {}
    virtual ~Animal() {}
    virtual void Speak() = 0;
    virtual const char* GetName() { return name_; }

protected:
    int         age_;
    const char* name_;
};

// dt WinDbgDemo!Dog
class Dog : public Animal {
public:
    Dog(int age, const char* name, const char* breed)
        : Animal(age, name), breed_(breed) {
    }

    void Speak() override {
        printf("Woof!\n");
    }

private:
    const char* breed_;
};

// ---------- 9. 全局变量 ----------
// dy / dt 观察全局变量符号
int         g_globalInt = 0x12345678;
const char* g_globalStr = "Global String in .rdata";
Point       g_globalPoint = { 0xAABB, 0xCCDD };

// ============================================================
// 主函数
// ============================================================
int main() {
    // ===== 基本类型 (db / dw / dd / dq 观察) =====
    char      c = 'A';                          // 0x41
    short     s = 0x1234;
    int       i = 0x12345678;
    long      l = 0x12345678;
    long long ll = 0x123456789ABCDEF0;
    float     f = 3.14f;
    double    d = 3.141592653589793;

    // ===== 指针变量 (dpp/dpu/dpa/dps 观察解引用链) =====
    int  intVar = 100;
    int* pInt = &intVar;

    // ===== 数组 (db / dw / dd 观察内存布局) =====
    int     intArray[8] = { 10, 20, 30, 40, 50, 60, 70, 80 };
    char    charArray[16] = "HelloWinDbg";        // da 观察字符串
    wchar_t wcharArray[16] = L"HelloWinDbg";      // du 观察宽字符串

    // ===== 字符串 (da / du / dS 观察) =====
    const char* str = "Hello WinDbg";
    const wchar_t* wstr = L"Hello WinDbg";
    std::string    cppstr = "C++ String";
    std::wstring   cppwstr = L"C++ WString";

    // ===== 结构体实例 (dt 观察字段布局) =====
    Point     pt = { 100, 200 };
    Buffer    buf = { "ABCDEFGHIJKLMNO", 15, 16 };
    Rectangle rect = { {10, 20}, {110, 220}, RED };

    // ===== 链表 (dpp/dps 观察指针解引用) =====
    Node n1 = { 1, nullptr, nullptr };
    Node n2 = { 2, nullptr, nullptr };
    Node n3 = { 3, nullptr, nullptr };
    n1.next = &n2;  n2.prev = &n1;
    n2.next = &n3;  n3.prev = &n2;

    // ===== 类 (vtable 在对象首字段, dpp 观察虚表) =====
    Dog dog(3, "Buddy", "Golden Retriever");

    // ===== 联合体 (dt 观察共享内存) =====
    Data data;
    data.i = 0x41424344;   // 同时是 'ABCD' (db) 和 float (df)

    // ===== 位域 (dt / db 观察位布局) =====
    Flags flags;
    flags.visible = 1;
    flags.enabled = 0;
    flags.paused = 1;
    flags.reserved = 0;
    flags.priority = 5;

    // ===== 枚举 =====
    Color color = BLUE;

    // ===== 打印地址 (方便在 WinDbg 中定位) =====
    printf("=== WinDbg d-Cmd Demo ===\n");
    printf("&c        = %p\n", &c);
    printf("&s        = %p\n", &s);
    printf("&i        = %p\n", &i);
    printf("&ll       = %p\n", &ll);
    printf("&f        = %p\n", &f);
    printf("&d        = %p\n", &d);
    printf("pInt      = %p  (target=%p)\n", pInt, (void*)*pInt);
    printf("intArray  = %p\n", intArray);
    printf("charArray = %p  (use da)\n", charArray);
    printf("wcharArray= %p  (use du)\n", wcharArray);
    printf("str       = %p  (use da)\n", str);
    printf("wstr      = %p  (use du)\n", wstr);
    printf("&pt       = %p  (dt Point)\n", &pt);
    printf("&buf      = %p  (dt Buffer)\n", &buf);
    printf("&rect     = %p  (dt Rectangle)\n", &rect);
    printf("&n1       = %p  (dt Node; dpp &n1+0x8)\n", &n1);
    printf("&dog      = %p  (dt Dog; vtable at *(&dog))\n", &dog);
    printf("&data     = %p  (dt Data)\n", &data);
    printf("&flags    = %p  (dt Flags)\n", &flags);
    printf("&color    = %p  (dt Color)\n", &color);
    printf("&g_globalInt   = %p\n", &g_globalInt);
    printf("&g_globalStr   = %p  (use da)\n", &g_globalStr);
    printf("&g_globalPoint = %p  (dt Point)\n", &g_globalPoint);

    // ===== 在这里断下, 此时所有局部变量均已初始化 =====
    // WinDbg 附加方式:
    //   1) 直接附加: windbg -pn Project1.exe
    //   2) 启动调试: windbg Project1.exe
    //   3) 生成 dump: .dump /f full.dmp
    printf("\nNow program will break, attach WinDbg or save dump...\n");
    __debugbreak();

    // 防止变量被优化掉
    (void)c; (void)s; (void)i; (void)l; (void)ll; (void)f; (void)d;
    (void)pInt; (void)intArray; (void)charArray; (void)wcharArray;
    (void)str; (void)wstr; (void)cppstr; (void)cppwstr;
    (void)pt; (void)buf; (void)rect; (void)n1; (void)n2; (void)n3;
    (void)dog; (void)data; (void)flags; (void)color;

    return 0;
}

## 内存转储（Memory Dump)

> **内存转储 = 程序在某一时刻的内存快照**

就像给程序拍了一张"照片"，记录下当时所有的内存状态。但注意：**只有程序处于"异常状态"时，这张照片才有分析价值**。

| 状态                 | 是否有价值 | 原因                  |
| -------------------- | ---------- | --------------------- |
| 崩溃/异常            | ✅ 高价值   | 能定位根因            |
| 挂起/死锁            | ✅ 高价值   | 能看到线程卡在何处    |
| 高 CPU 占用          | ✅ 高价值   | 能分析热点代码        |
| 正常等待（消息循环） | ❌ 低价值   | 只是 idle，无异常信息 |

用户模式 vs 内核模式转储的区别

| 类型             | 包含内容                                     | 类比           |
| ---------------- | -------------------------------------------- | -------------- |
| **用户模式转储** | 单个进程的 **已提交字节（Committed Bytes）** | 一个房间的内部 |
| **内核模式转储** | **整个物理内存（RAM）** 的副本 + 头部元数据  | 整栋楼的全景   |

已提交字节（Committed Bytes）是什么？意思是程序能合法访问的所有内存。包括：

- `VirtualAlloc` / `HeapAlloc` / `malloc` 分配的内存
- `new` 操作符分配的堆内存
- 全局变量占用的内存
- 已加载的 DLL 代码和数据段

转储文件的类型对比

| 类型                      | 包含信息                                     | 体积               | 适用场景                     |
| ------------------------- | -------------------------------------------- | ------------------ | ---------------------------- |
| **完整转储（Full Dump）** | 进程所有已提交内存、堆、栈、线程、模块、句柄 | 很大（可能 GB 级） | 深度分析、首次异常           |
| **小型转储（Mini Dump）** | 仅调用栈、寄存器、部分内存页                 | 很小（KB~MB 级）   | 频繁异常、线上收集、存储受限 |

转储与二进制分析的区别：二进制是纯静态的，存储的话是动态的某一时刻

| 对比项           | 转储文件 | 二进制文件（PE） |
| ---------------- | -------- | ---------------- |
| 包含代码         | ✅        | ✅                |
| 包含运行时堆/栈  | ✅        | ❌                |
| 包含线程状态     | ✅        | ❌                |
| 包含寄存器值     | ✅        | ❌                |
| 包含内存分配信息 | ✅        | ❌                |

调试方式的差异性

| 调试方式                        | 能力                         | 限制                           |
| ------------------------------- | ---------------------------- | ------------------------------ |
| **转储调试**                    | 分析单一时刻的状态           | ❌ 只能看，不能执行（像看照片） |
| **动态调试（Live）**            | 单步执行、下断点、观察变化   | ✅ 能前进，不能后退             |
| **时间旅行调试（TTD）**新版才有 | 能前进 + 能后退（`g-` 回溯） | 需要特殊录制，体积大           |

## 调试符号

**调试符号 = 二进制地址 ↔ 人类可读名称的映射表**

**类比：** 符号就像 **DNS** —— 把难记的 IP 地址（二进制地址）解析成域名（函数名/变量名）。

| 层面       | 没有符号                  | 有符号                                  |
| ---------- | ------------------------- | --------------------------------------- |
| 看到的代码 | `0x401000` `0x401005`     | `main` `printf`                         |
| 看到的变量 | `[ebp+8]` `[esp+4]`       | `argc` `p`                              |
| 调用栈     | `0x77a12340` → `0x401020` | `kernel32!BaseThreadInitThunk` → `main` |
| 源码定位   | 无                        | `hello.cpp:15`                          |

  这种就是有符号的情况，能看到那些信息，如果没符号那就是纯二进制了很苦涩难懂

流程图如下：

源代码 (.cpp)          编译器              目标文件 (.obj)
   main.cpp    ──────►   cl.exe   ──────►   main.obj
   utils.cpp   ──────►   cl.exe   ──────►   utils.obj
                              │
                              ▼
                         链接器 (link.exe)
                              │
                              ▼
                    ┌─────────────────┐
                    │  可执行文件 (.exe)  │  ← 只有机器码

                    └─────────────────┘
```

| 阶段 | 工具       | 产物   | 包含符号？         |
| ---- | ---------- | ------ | ------------------ |
| 编译 | `cl.exe`   | `.obj` | 部分（未解析）     |
| 链接 | `link.exe` | `.exe` | ❌ 无（只有机器码） |
| 链接 | `link.exe` | `.pdb` | ✅ 全部符号信息     |

.pdb 文件里到底有什么？

| 信息类型              | 说明                           |
| --------------------- | ------------------------------ |
| **函数名 ↔ 地址映射** | `main` = `0x401000`            |
| **变量名 ↔ 内存位置** | `p` = `[ebp-4]`                |
| **源码路径**          | `C:\project\hello.cpp`         |
| **行号信息**          | `hello.cpp:15` 对应 `0x401005` |
| **栈帧大小**          | 函数局部变量占用的栈空间       |
| **优化信息**          | 内联、寄存器分配等             |
| **类型信息**          | 结构体、类定义                 |

旧.pdb是不能用的，原因如下：

| 情况                                    | 结果           | 原因                                 |
| --------------------------------------- | -------------- | ------------------------------------ |
| 同一源码，重新编译                      | ❌ 旧 .pdb 失效 | 编译器可能分配不同地址、优化方式不同 |
| 复制 .exe 到新机器，没复制 .pdb         | ⚠️ 调试器无符号 | 符号路径配置问题                     |
| 使用 Release 的 .pdb 调试 Debug 的 .exe | ❌ 完全不匹配   | 编译选项不同，地址完全不同           |



**Windbg 会检查：**时间戳（Timestamp），文件大小，GUID / Signature

不匹配时，Windbg 会提示 `*** ERROR: Symbol file could not be found.` 或加载但显示 `MISMATCHED`。

## 调用栈、全局变量和局部变量

| 概念              | 一句话                                                 |
| ----------------- | ------------------------------------------------------ |
| **调用栈**        | 函数调用链，调试第一信息。`k` 命令查看。               |
| **线程 (Thread)** | 词源来自希腊神话"线团"——忒修斯用线导航迷宫。           |
| **局部变量**      | 第二重要，但64位优化可能放寄存器里，`dv` 看不到。      |
| **全局变量**      | 始终可用，固定地址，不受优化影响。                     |
| **调试优先级**    | `k` (调用栈) → `dv` (局部变量) → `x`/`r` (全局/寄存器) |

  比较容易的混淆点就是这里了。**调用栈是"逻辑上的函数调用关系"，线程栈是"物理上的内存布局"**。两者描述的是同一个东西的不同维度。

| 调用栈 (Call Stack)         | 线程栈 (Thread Stack)           |
| --------------------------- | ------------------------------- |
| **逻辑概念** — 函数调用关系 | **物理内存** — 一块连续的栈空间 |

```
线程栈（内存区域）              调用栈（逻辑链）
┌─────────────────┐
│  高地址          │
│  ┌─────────┐    │     ┌─────────┐
│  │ 栈帧N   │    │────→│ funcC() │ 栈顶
│  │ (funcC) │    │     └────┬────┘
│  ├─────────┤    │     ┌────┴────┐
│  │ 栈帧N-1 │    │────→│ funcB() │
│  │ (funcB) │    │     └────┬────┘
│  ├─────────┤    │     ┌────┴────┐
│  │ 栈帧N-2 │    │────→│ funcA() │
│  │ (funcA) │    │     └────┬────┘
│  ├─────────┤    │     ┌────┴────┐
│  │ 栈帧1   │    │────→│ main()  │ 栈底
│  │ (main)  │    │     └─────────┘
│  └─────────┘    │
│  低地址          │
└─────────────────┘
        ↑
    ESP/RSP
```



相关命令如下：

```
k          # 查看调用栈
dv         # 查看局部变量（可能优化不可见）
x mod!g_*  # 查看全局变量
r          # 查看寄存器
```

---

二、进程 vs 线程：谁才是真正干活的？

| 实体     | 本质                      | 是否执行代码         | 类比                   |
| -------- | ------------------------- | -------------------- | ---------------------- |
| **进程** | 内存地址空间的容器        | ❌ **不执行任何指令** | 房子（有地址、有房间） |
| **线程** | CPU 指令序列 + 执行上下文 | ✅ **真正的执行单元** | 房子里的人（实际活动） |

### 进程 = "家"（内存容器）
- 提供独立的虚拟地址空间
- 包含代码段、数据段、堆、线程栈
- **进程本身不跑代码，只是给线程提供运行环境**

### 线程 = "干活的人"（执行单元）
- 一组按顺序执行的 CPU 指令
- 拥有独立的 **调用栈**（存储局部变量、返回地址）
- 拥有 **CPU 上下文**（寄存器值的快照）
- 线程切换时，操作系统保存/恢复寄存器状态

---

内核态 vs 用户态

| 特性         | 内核模式 (Kernel Mode)                 | 用户模式 (User Mode)     |
| ------------ | -------------------------------------- | ------------------------ |
| **权限**     | 最高特权 (Ring 0)                      | 受限特权 (Ring 3)        |
| **内存访问** | 可访问所有物理内存                     | 只能访问本进程地址空间   |
| **硬件访问** | 直接访问磁盘、显卡、网卡等             | 必须通过系统调用间接访问 |
| **指令执行** | 可执行特权指令 (如 `in`, `out`, `cli`) | 执行特权指令 → 触发异常  |
| **运行实体** | 内核线程、驱动线程                     | 应用程序线程             |
| **栈**       | 内核栈                                 | 用户栈                   |

关键隔离机制

```
用户模式 Notepad 线程
    │
    ▼ 尝试读取 Excel 进程内存
    ┌─────────────────┐
    │  CPU 检查页表   │  ← 页表项标记该内存属于 Excel
    │  权限不足！      │
    │  触发 #PF 异常   │  ← Page Fault / Access Violation
    └─────────────────┘
    │
    ▼ 进入内核异常处理
    操作系统决定：终止 Notepad 或返回错误码
```

> **内存隔离是用户模式的核心安全特性** —— 没有它，一个恶意程序就能读取所有其他进程的数据。

---

线程栈：为什么每个线程都要有独立的栈？

笔记里提到栈的功能，这是理解线程的关键：

```
线程栈结构（从高地址向低地址增长）

高地址 ┌─────────────────┐
      │   返回地址        │  ← 函数调用时压入
      │   局部变量        │  ← 函数内声明的变量
      │   参数            │  ← 传递给函数的参数
      │   保存的寄存器     │  ← 调用约定要求保存的寄存器
      │   ...            │
      ├─────────────────┤ ← 栈顶 (ESP/RSP)
      │   未使用空间      │
低地址 └─────────────────┘
```

### 栈的核心作用

| 功能             | 说明                                         |
| ---------------- | -------------------------------------------- |
| **维护执行顺序** | 函数调用链：`main()` → `funcA()` → `funcB()` |
| **存储局部变量** | `int a = 10;` 存在栈上                       |
| **保存返回地址** | `call` 指令把下一条地址压栈，`ret` 时弹出    |
| **传递参数**     | 通过栈或寄存器传参                           |

### 用户栈 vs 内核栈

| 类型       | 所属         | 用途                       | 大小                         |
| ---------- | ------------ | -------------------------- | ---------------------------- |
| **用户栈** | 用户模式线程 | 用户代码执行               | 通常 1MB（可配置）           |
| **内核栈** | 内核模式线程 | 内核代码执行、系统调用处理 | 通常 12KB (x86) / 24KB (x64) |

> **重要：** 当线程通过系统调用进入内核时，会切换到该线程对应的 **内核栈**，用户栈指针被保存。

---

异常与中断：只发生在单个线程上下文

笔记提到这一点，非常关键：

```
时间线 ─────────────────────────────────────────────►

线程A 执行中 → 触发除零异常 → CPU 保存线程A上下文
                                              │
                                              ▼
                                    进入内核异常处理程序
                                    使用线程A的内核栈
                                              │
                                              ▼
                                    处理完成后返回线程A
                                    或终止线程A（未处理异常）

线程B 不受影响 ←──────────────────────────────────────
```

> **异常（中断）永远不会跨线程传播**，它只发生在触发它的那个线程上下文中。

---

六、内核调试视角的特殊性

### 为什么内核调试和进程调试不同？

| 调试类型         | 目标     | 看到的"进程"                                         |
| ---------------- | -------- | ---------------------------------------------------- |
| **用户模式调试** | 单个进程 | 该进程的线程、内存、模块                             |
| **内核模式调试** | 整个系统 | 内核本身就像一个"大进程"，所有驱动、系统线程都在里面 |

内核调试时看到的线程

```windbg
# 内核调试器中查看所有线程
kd> !process 0 0          # 列出所有进程
kd> !process <EPROCESS> 7  # 查看某个进程的详细信息（含线程）
kd> !thread <ETHREAD>      # 查看线程详情
```

在内核调试器中，你能看到：
- 系统进程 (`System`) 的线程 —— 内核本身的线程
- 各个驱动程序的线程
- 所有用户进程的线程（但它们在用户态执行时，内核调试器看到的是它们的内核栈）



| 重点                      | 核心内容                                               |
| ------------------------- | ------------------------------------------------------ |
| **进程不执行代码**        | 进程是内存容器，线程才是执行单元                       |
| **内核 = 大型进程**       | 包含驱动、系统组件，运行在特权模式                     |
| **内存隔离**              | 用户模式线程只能访问本进程内存，内核线程可访问所有内存 |
| **线程切换 = 寄存器切换** | 保存旧线程上下文，加载新线程上下文                     |
| **异常单线程性**          | 异常只发生在单个线程上下文，不会扩散                   |
| **双栈结构**              | 用户线程有用户栈，进入内核时切换到内核栈               |



## 调试器基础概念

| 状态        | 表现                   | 能否执行命令       |
| ----------- | ---------------------- | ------------------ |
| **Running** | 程序在跑，命令窗口灰掉 | ❌ 大部分命令禁用   |
| **Broken**  | 程序被中断/冻结        | ✅ 可以执行任意命令 |

附加到进程时，调试器会**冻结**目标，进入 Broken 状态。

Dump类型如下：

| 类型            | 触发原因                              | 包含内容                    |
| --------------- | ------------------------------------- | --------------------------- |
| **Crash Dump**  | 未处理异常（Second Chance），程序崩溃 | 崩溃时刻的进程状态          |
| **Hang Dump**   | 程序/系统极慢、无响应                 | 卡死时刻的状态              |
| **User Dump**   | 用户模式进程快照                      | 单个进程内存                |
| **Kernel Dump** | 系统级快照                            | **整个物理内存** + 所有进程 |



跟时间旅行调试的对比，这个我们再内存存储讲过

|          | 普通 Dump | TTT                  |
| -------- | --------- | -------------------- |
| 状态     | 单点快照  | 完整执行记录         |
| 能否回溯 | ❌ 不能    | ✅ 可以前后跳转       |
| 工具     | Windbg    | WinDbg Preview / TTD |

内存转储的命令如下：

| 命令        | 类型         | 体积 | 包含内容                               |
| ----------- | ------------ | ---- | -------------------------------------- |
| `.dump /m`  | 基础迷你转储 | 最小 | 仅基本信息                             |

#### 讲解r和k

  **寄存器查看（`r`）和调用栈显示（`k`）** 两个核心命令，是调试中最常用的操作。

  先讲r命令：

| 命令          | 作用             |
| ------------- | ---------------- |
| `r`           | 显示所有寄存器   |
| `r rax`       | 查看单个寄存器   |
| `r rax=0x100` | **修改**寄存器值 |
| `r zf`        | 查看标志位       |

有个特性是：

```
r 显示的是"当前线程上下文"的寄存器
    │
    ▼ 切换线程后自动更新
~1s          # 切换到线程1
r            # 显示的是线程1的寄存器
```

**不是 CPU 物理寄存器，而是线程上下文的快照**（线程切换时保存/恢复的寄存器值）

查看所有线程用~  切换至id为1的线程需要使用：~1s。

![image-20260717002231524](/img/windbg笔记1/image-20260717002231524.png)

这里分析是这样的：

| 线程         | 状态                            | 说明                         |
| ------------ | ------------------------------- | ---------------------------- |
| **线程 0**   | `Project1!wmain`                | 主线程，执行你的代码         |
| **线程 1-3** | `NtWaitForWorkViaWorkerFactory` | 线程池工作线程，**空闲等待** |

常见的多线程命令如下：

```
~              # 列出所有线程
~1s            # 切换到线程1
~1 k           # 查看线程1的调用栈（不切换）
~* k           # 查看所有线程的调用栈
~1 r           # 查看线程1的寄存器（不切换）
~1 f           # 冻结线程1（不让它运行）
~1 u           # 解冻线程1
```



k命令如下：

| 命令 | 显示内容          | 用途         |
| ---- | ----------------- | ------------ |
| `k`  | 基础调用栈        | 最常用       |
| `kb` | + 前三个参数      | 快速看参数   |
| `kv` | + 参数 + FPO 信息 | 详细分析     |
| `kp` | + 参数类型和名称  | 需要完整符号 |
| `kc` | 只显示函数名      | 简洁视图     |
| `kf` | + 栈帧大小        | 内存分析     |

![image-20260717005940659](/img/windbg笔记1/image-20260717005940659.png)

我们解读一下最详细的命令kp的值：

| 字段            | 含义                                 | 示例值                  |
| --------------- | ------------------------------------ | ----------------------- |
| `00`            | 栈帧编号（0=当前，1=调用者...）      | `00`                    |
| `Child-SP`      | **栈帧基址**（当前函数的栈起始地址） | `0000003d`ed39f550\`    |
| `RetAddr`       | **返回地址**（函数执行完跳回哪里）   | `00007ff7`e3af22f9\`    |
| `Args to Child` | 传给下一层函数的参数（前3个）        | `0000d7fa`00000001\`... |

地址格式 0000003d`ed39f550 这个反引号是隔离高32位和低32位的。我们这64位程序实际就是读就是：

 `0x0000003ded39f550`

Args to child的值以上面截图为案例：

| 值                              | 含义                                               |
| ------------------------------- | -------------------------------------------------- |
| `Child-SP = 0000003d`ed39f550\` | `wmain` 函数的栈帧从这里开始                       |
| `RetAddr = 00007ff7`e3af22f9\`  | `wmain` 跑完回到 `invoke_main+0x39`                |
| `Args[0] = 0000d7fa`00000001\`  | 参数1（`argc` = 1，但这里看起来是垃圾值/未初始化） |
| `Args[1] = 0000018e`235194f0\`  | 参数2（`argv` 指针 = `0x0000018e235194f0`）        |
| `Args[2] = 00000000`00000000\`  | 参数3（未使用）                                    |
| `Args[3] = 00007ff7`e3af3a2d\`  | 额外显示的值                                       |

**64位 x64 调用约定用寄存器传参**，`Args to Child` 显示的是栈上的值，**不一定等于实际参数**。`kp` 显示的参数更准确。

kp里的call site更详细：

![image-20260717010847013](/img/windbg笔记1/image-20260717010847013.png)

```
Project1!wmain(int argc = 0n1, wchar_t ** argv = 0x0000018e`235194f0)+0x40 [test.cpp @ 12]
├──────┤├────┤├──────────────────────────────────────────────────────┤├────┤├────────────┤
 模块名  函数名            参数类型和值（需要私有符号）                    偏移    源码位置
```

这样理解就行了。。

| 重点                   | 核心内容                       |
| ---------------------- | ------------------------------ |
| `r` 显示的是线程上下文 | 不是物理 CPU，切换线程后值会变 |
| `r` 可以修改寄存器     | 直接改执行流，非常危险         |
| `kv` 参数显示不可靠    | 64位和优化代码中参数可能不准   |
| `~*k` 查看所有线程     | 多线程死锁分析必备             |
| FPO 优化影响栈解析     | EBP 被优化，调用栈可能断裂     |



#### 讲解e和u

讲的是 **内存编辑和反汇编命令**，是实时调试中修改程序行为的核心操作

核心限制

>
> Dump 是静态快照，没有可写的内存。

e的常用命令如下：

| 命令                  | 作用                                    | 示例                     |
| --------------------- | --------------------------------------- | ------------------------ |

| `eza 地址 "字符串"`   | 写入 ASCII 字符串（**以 NULL 结尾**）   | `eza 0x277b30 "hello"`   |
| `eu 地址 "字符串"`    | 写入 Unicode 字符串（不以 NULL 结尾）   | `eu 0x277b30 "你好"`     |
| `ezu 地址 "字符串"`   | 写入 Unicode 字符串（以 NULL 结尾）     | `ezu 0x277b30 "world"`   |
| `eb 地址 值1 值2 ...` | 写入字节（Byte）                        | `eb 0x277b30 41 42 43`   |

##### 反汇编命令（`u` / `ub` / `uf`）

1. `u` — 正向反汇编

```windbg
0:000> u                          # 从当前 EIP 开始反汇编
0:000> u eip L20                  # 从 EIP 开始，反汇编 20 字节
0:000> u 0x401000 L30             # 从指定地址开始，反汇编 30 字节
```

2. `ub` — 反向反汇编

```windbg
0:000> ub eip L10                 # 从 EIP 往前反汇编 10 字节
0:000> ub 0x401020                # 从指定地址往前反汇编
```

> **用于回溯**：知道当前位置，想看前面执行了什么指令。

3. `uf` — 反汇编整个函数

```windbg
0:000> uf Project1!wmain          # 反汇编 wmain 整个函数
0:000> uf /c Project1!wmain       # 只显示函数中的外部调用（call）
```

| 参数    | 作用                             |
| :------ | :------------------------------- |
| `uf`    | 完整反汇编函数                   |
| `uf /c` | 只显示 **call 指令**（外部调用） |

> `uf /c` 在逆向分析中很有用，快速看函数调用了哪些 API。

##### 内核模式调试手动遍历 Windows 内核进程链表

内核调试参考下面章节的配置

```
; 1. 查 EPROCESS 关键字段偏移 确认 ActiveProcessLinks、UniqueProcessId、ImageFileName 在 _EPROCESS 结构体中的偏移量。
dt nt_exe!_EPROCESS -y ActiveProcessLinks UniqueProcessId ImageFileName

; 2. 查找全局变量 PsInitialSystemProcess 的地址
PsInitialSystemProcess 是一个 指针变量，指向 System 进程的 _EPROCESS 结构体。
x nt_exe!PsInitialSystemProcess
; → fffff800`29d63028

; 3. 读这个指针的值（解引用），拿到 System 进程的 EPROCESS 地址
dq nt_exe!PsInitialSystemProcess L1
; → ffffe000`1ee70900   (这就是 System 进程的 EPROCESS)

; 4. 查看 System 进程
dt nt_exe!_EPROCESS ffffe0001ee70900 -y UniqueProcessId ImageFileName ActiveProcessLinks

; 5. 遍历整条进程链表（核心命令）
dt nt_exe!_EPROCESS -l ActiveProcessLinks.Flink -y Image* ffffe0001ee70900
```

## **32位汇编栈帧分析与调用约定**

```
高地址 ↑
       │  参数传递区（Arguments）    ← 调用者传入
       │  返回地址（Return Address） ← 函数执行完跳回这里
       │  帧指针 EBP（Frame Pointer）← 定位栈帧基址
       │  局部变量（Local Variables）← 函数内部变量
       ↓
低地址
```

**栈向下增长**：高地址 → 低地址，栈顶地址最小。

| 概念         | 说明                                             |
| ------------ | ------------------------------------------------ |
| **栈帧**     | 每个函数调用在栈上的独立片段                     |
| **EBP**      | 帧指针，标记栈帧基址（优化可省略）               |
| **返回地址** | `ret` 指令跳转的目标                             |
| **调用约定** | 决定参数怎么传、谁清理栈（`cdecl`/`stdcall` 等） |

栈帧结构（32位）**着重理解这个结构！**

```plain
高地址 ↑
       │  参数（AP）      ← EBP+8、+0xC、+0x10...
       │  返回地址（RA）  ← EBP+4
       │  保存的 EBP      ← EBP（当前栈帧基址）
       │  局部变量（LV）   ← EBP-4、-8、-0xC...
       ↓
低地址
```

**函数序言**（开头）：

```asm
push ebp          ; 保存旧 EBP
mov ebp, esp      ; EBP 指向当前栈顶
sub esp, 0x10     ; 分配局部变量空间
```

**函数尾声**（结尾）：

```asm
mov esp, ebp      ; 恢复栈指针
pop ebp           ; 恢复旧 EBP
ret               ; 弹出返回地址，跳转
```

------

2. CALL 指令做了什么

| 步骤 | 操作                                       |
| :--- | :----------------------------------------- |
| 1    | `ESP -= 4`，压入返回地址（下一条指令地址） |
| 2    | `EIP = 目标函数地址`，跳转                 |

**RET 是逆操作**：弹出栈顶到 EIP，ESP += 4。

3. 调用约定对比

| 约定          | 谁清理栈           | 特点                       | 示例                 |
| :------------ | :----------------- | :------------------------- | :------------------- |
| **__cdecl**   | 调用方（Caller）   | 支持可变参数               | C/C++ 默认、`printf` |
| **__stdcall** | 被调用方（Callee） | 代码体积小，不支持可变参数 | Windows API、`ret 8` |

**__cdecl** 调用后：

```asm
add esp, 0Ch      ; 调用方清理 3 个参数（12 字节）
```

**__stdcall** 函数末尾：

```asm
ret 8             ; 被调用方清理 2 个参数（8 字节）
```

### 实操

  这是原代码

```
// 栈帧分析实验（32位 Debug 版）
// 编译：用 VS2022 "x86 Native Tools Command Prompt" 执行 cl /Zi /Od /MTd stackframe_demo.cpp
// 或在 VS2022 里新建 Win32 控制台项目（x86 Debug），把这个文件加进去编译
//
// 实验目标：
//   1. 观察栈帧结构：参数 → 返回地址 → EBP → 局部变量
//   2. 观察 CALL 指令压入返回地址
//   3. 对比 __cdecl（调用方清栈）vs __stdcall（被调用方清栈）

#include <cstdio>
#include <windows.h>

// ========== __cdecl 函数：调用方清栈 ==========
// 编译器生成的序言会是：
//   push ebp
//   mov  ebp, esp
//   sub  esp, 0x??    ; 分配局部变量
// 结尾会是：
//   mov  esp, ebp
//   pop  ebp
//   ret               ; 注意：没有数字，调用方负责清栈
int __cdecl CdeclFunc(int a, int b, int c) {
    int local1 = a + b;       // 局部变量1，在 [ebp-4]
    int local2 = b + c;       // 局部变量2，在 [ebp-8]
    int local3 = a * c;       // 局部变量3，在 [ebp-0xC]
    return local1 + local2 + local3;
}

// ========== __stdcall 函数：被调用方清栈 ==========
// 结尾会是：
//   ret  0Ch          ; 注意：有 0Ch（12字节=3个int参数），被调用方清栈
int __stdcall StdcallFunc(int a, int b, int c) {
    int local1 = a * b;
    int local2 = b * c;
    int local3 = a * c;
    return local1 + local2 + local3;
}

// ========== 一个带更多局部变量的函数，方便观察栈帧大小 ==========
int __cdecl ComplexFunc(int a, int b, int c, int d) {
    int arr[4] = { a, b, c, d };   // 数组在栈上
    int sum = 0;
    for (int i = 0; i < 4; i++) {
        sum += arr[i];
    }
    return sum;
}

int main() {
    // 这3个变量在 main 的栈帧里
    int x = 10;
    int y = 20;
    int z = 30;

    printf("==== Stack Frame Demo (32-bit) ====\n");
    printf("&x = %p, &y = %p, &z = %p\n", &x, &y, &z);
    printf("&CdeclFunc   = %p\n", (void*)CdeclFunc);
    printf("&StdcallFunc = %p\n", (void*)StdcallFunc);
    printf("&ComplexFunc = %p\n", (void*)ComplexFunc);
    printf("===================================\n");

    // 在调用前断下，观察调用栈布局
    __debugbreak();

    // 第1次调用 __cdecl 函数
    int r1 = CdeclFunc(x, y, z);
    printf("CdeclFunc(%d, %d, %d) = %d\n", x, y, z, r1);

    __debugbreak();

    // 第2次调用 __stdcall 函数
    int r2 = StdcallFunc(x, y, z);
    printf("StdcallFunc(%d, %d, %d) = %d\n", x, y, z, r2);

    __debugbreak();

    // 第3次调用带数组的函数
    int r3 = ComplexFunc(x, y, z, 40);
    printf("ComplexFunc(%d, %d, %d, %d) = %d\n", x, y, z, 40, r3);

    printf("\n==== 实验结束，可在 WinDbg 中练习 ====\n");
    __debugbreak();

    return 0;
}

```

### 调用约定汇总

| 对比项          | `__cdecl`              | `__stdcall`            | `__fastcall`           | `__thiscall` (C++)        |
| --------------- | ---------------------- | ---------------------- | ---------------------- | ------------------------- |
| **参数传递**    | 全部压栈（从右到左）   | 全部压栈（从右到左）   | ECX/EDX + 其余压栈     | ECX = this 指针，其余压栈 |
| **清栈方**      | **调用方（Caller）**   | **被调用方（Callee）** | **被调用方（Callee）** | **被调用方（Callee）**    |
| **ret 指令**    | `ret`（无数字）        | `ret N`（有数字）      | `ret N`（有数字）      | `ret N`（有数字）         |
| **可变参数**    | ✅ **支持**             | ❌ 不支持               | ❌ 不支持               | ❌ 不支持                  |
| **代码体积**    | 大（每个调用点都清栈） | 小（集中清栈）         | 小                     | 小                        |
| **性能**        | 一般                   | 一般                   | 稍好（寄存器传参）     | 一般                      |
| **C 默认**      | ✅ 是                   | ❌ 否                   | ❌ 否                   | —                         |
| **C++ 默认**    | ✅ 是（非成员）         | ❌ 否                   | ❌ 否                   | ✅ 是（成员函数）          |
| **Windows API** | ❌ 否                   | ✅ **是**（`WINAPI`）   | ❌ 否                   | ❌ 否                      |
| **函数名修饰**  | `_func`                | `_func@12`             | `@func@12`             | 复杂（含 this 信息）      |



## 64位汇编栈帧分析与调用约定

  相比之下64位的汇编调用约定上是统一的。

|          | 32位                                       | 64位                                   |
| -------- | ------------------------------------------ | -------------------------------------- |
| 调用约定 | 多种（`__cdecl`/`__stdcall`/`__fastcall`） | **只有一种**（x64 calling convention） |
| 参数传递 | 全部压栈                                   | 前4个用**寄存器**，第5个起压栈         |
| 清栈     | 争论谁清（cdecl vs stdcall）               | **调用者清栈**，无争议                 |

寄存器对比大全

| 64位    | 32位 | 16位 | 8位   | 说明                   |
| ------- | ---- | ---- | ----- | ---------------------- |
| **RAX** | EAX  | AX   | AL/AH | 累加器，返回值         |
| **RBX** | EBX  | BX   | BL/BH | 基址寄存器             |
| **RCX** | ECX  | CX   | CL/CH | 计数器，第1参数        |
| **RDX** | EDX  | DX   | DL/DH | 数据寄存器，第2参数    |
| **RSI** | ESI  | SI   | SIL   | 源索引                 |
| **RDI** | EDI  | DI   | DIL   | 目的索引               |
| **RBP** | EBP  | BP   | BPL   | 基址指针（64位很少用） |
| **RSP** | ESP  | SP   | SPL   | 栈指针                 |
| **R8**  | R8D  | R8W  | R8B   | 新增，第3参数          |
| **R9**  | R9D  | R9W  | R9B   | 新增，第4参数          |
| **R10** | R10D | R10W | R10B  | 新增                   |
| **R11** | R11D | R11W | R11B  | 新增                   |
| **R12** | R12D | R12W | R12B  | 新增                   |
| **R13** | R13D | R13W | R13B  | 新增                   |
| **R14** | R14D | R14W | R14B  | 新增                   |
| **R15** | R15D | R15W | R15B  | 新增                   |

 这个R8到R15 咋理解？**64位新增的通用寄存器**，作用就是**多出来的变量存储空间**。其中R8和R9是固定做第三参数和第四参数，而R10-R15就比较灵活了，由于没有固定用途，编译器随便用。

64位上前4个用**寄存器**，第5个起压栈。区别就是前4个参数不用 push/pop，寄存器访问比内存快得多

```
mov ecx, 1      ; 第1个参数 → RCX
mov edx, 2      ; 第2个参数 → RDX
mov r8d, 3      ; 第3个参数 → R8
mov r9d, 4      ; 第4个参数 → R9
push 5          ; 第5个参数 → 栈
call func
```



我们知道32位是靠EBP 固定锚点来定位的

- EBP 是**固定锚点**，整个函数期间不变
- 参数在 EBP **上方**（+偏移），局部变量在 EBP **下方**（-偏移）
- 调试友好：`k` 命令靠 EBP 链回溯调用栈

```
高地址 ↑
       │  参数 c      [ebp+0x10]  ← EBP + 正偏移 = 参数
       │  参数 b      [ebp+0x0C]
       │  参数 a      [ebp+0x08]
       │  返回地址    [ebp+0x04]
       │  旧 EBP      [ebp]      ← EBP 固定指向这里（不动！）
       │  ─────────────
       │  局部变量1   [ebp-0x04]  ← EBP - 负偏移 = 局部变量
       │  局部变量2   [ebp-0x08]
       ↓
低地址
```



而64位全靠RSP

- **没有 EBP**，所有东西相对 RSP
- RSP 是**动态的**，push/pop/sub/add 都会变
- 全部用**正偏移**（RSP+偏移），因为栈向下增长，已分配的内存在 RSP 上方，RSP和ESP下面的地址都是未分配的内存都不能考虑。

为什么ESP有负偏移是根据EBP这个固定描点变化的。

```
高地址 ↑
       │  [rsp+0x60]  局部变量3
       │  [rsp+0x58]  局部变量2
       │  [rsp+0x50]  局部变量1
       │  [rsp+0x48]  参数5（压栈）
       │  [rsp+0x40]  参数4副本（R9）
       │  [rsp+0x38]  参数3副本（R8）
       │  [rsp+0x30]  参数2副本（RDX）
       │  [rsp+0x28]  参数1副本（RCX）  ← 参数驻留区
       │  [rsp+0x20]  阴影空间（Shadow Space，固定0x20字节）
       │  [rsp+0x18]  ...
       │  [rsp+0x10]  ...
       │  [rsp+0x08]  返回地址
       │  [rsp]       ← RSP 指向这里（动态变化！）
       ↓
低地址
```



### x64 calling convention

 x64 calling convention 是 **64位 Windows 唯一的标准调用约定**，规定了函数参数怎么传、栈怎么分配、谁清理栈。

  参数传递刚才说了。

| 参数序号    | 传递方式             |
| ----------- | -------------------- |
| 第1个       | **RCX**              |
| 第2个       | **RDX**              |
| 第3个       | **R8**               |
| 第4个       | **R9**               |
| 第5个及以后 | **压栈**（从右到左） |

栈分配：阴影空间（Shadow Space）

```
sub rsp, 0x28       ; 至少分配 32 字节（0x20）+ 8 字节对齐 = 0x28
```

| 用途                                  | 大小                     |
| ------------------------------------- | ------------------------ |
| 阴影空间（Shadow Space / Home Space） | 0x20（32字节）           |
| 给被调用函数保存 RCX/RDX/R8/R9 用     |                          |
| 额外空间                              | 8 字节（16字节对齐要求） |

**即使被调用函数不用，调用方也必须分配**

 最后的话是由调用方来清栈

demo理解：

int func(int a, int b, int c, int d, int e);  // 调用 func(1, 2, 3, 4, 5)

```
; 调用方（main）
sub rsp, 0x28           ; 分配阴影空间 + 对齐

mov ecx, 1              ; 第1参数
mov edx, 2              ; 第2参数
mov r8d, 3              ; 第3参数
mov r9d, 4              ; 第4参数

push 5                  ; 第5参数压栈
call func

add rsp, 0x30           ; 清理：0x28 阴影空间 + 8 字节第5参数
```

```
这是被调用方的模板

func:
    ; 序言
    mov [rsp+0x08], rcx     ; 参数驻留（可选，Debug版有）
    mov [rsp+0x10], rdx
    mov [rsp+0x18], r8
    mov [rsp+0x20], r9
    
    ; 函数体
    mov eax, ecx            ; 用 RCX（第1参数）
    add eax, edx            ; 用 RDX（第2参数）
    
    ; 返回值在 RAX
    
    ; 尾声（无 add rsp，因为阴影空间是调用方分配的）
    ret
```

### 参数驻留

  刚才的案例提了一下

参数驻留（Parameter Homing）是 **64位 Debug 版特有的机制**，解决"寄存器参数容易被覆盖"的问题。

问题：寄存器参数会丢失

```asm
; 函数 A 调用函数 B(100, 200, 300, 400)
mov ecx, 100        ; 第1参数
mov edx, 200        ; 第2参数
mov r8d, 300        ; 第3参数
mov r9d, 400        ; 第4参数
call B
```

进入 B 后，ECX 里的 100 可能被覆盖：

```asm
B:
    mov ecx, 0      ; ECX 被清零！原来的 100 丢了
    ...
```

**调试时断在 B 中间，想看"谁传的 100"，但 ECX 早不是 100 了。**

---

解决方案：参数驻留

Debug 版编译器在函数开头，把寄存器参数**复制到栈上**：

```asm
B:
    ; 函数序言
    sub rsp, 0x28
    
    ; 参数驻留（把寄存器存到栈上）
    mov [rsp+0x08], rcx     ; 存第1参数
    mov [rsp+0x10], rdx     ; 存第2参数
    mov [rsp+0x18], r8      ; 存第3参数
    mov [rsp+0x20], r9      ; 存第4参数
    
    ; 后续代码随便改寄存器，栈上的副本还在
    mov ecx, 0              ; ECX 变了，但 [rsp+0x08] 还是 100
```

栈布局（有参数驻留）

```
高地址 ↑
       │  [rsp+0x28]  参数4副本 = 400
       │  [rsp+0x20]  参数3副本 = 300
       │  [rsp+0x18]  参数2副本 = 200
       │  [rsp+0x10]  参数1副本 = 100  ← 驻留区
       │  [rsp+0x08]  返回地址
       │  [rsp]       ← RSP 指向这里
       ↓
```

**调试器通过 `[rsp+0x10]` 等偏移，能看到原始参数值。**

---

Debug vs Release 对比

|            | Debug 版           | Release 版           |
| ---------- | ------------------ | -------------------- |
| 参数驻留   | ✅ 有               | ❌ 无                 |
| 栈空间     | 大（存了参数副本） | 小                   |
| 性能       | 慢（多几条 mov）   | 快                   |
| 调试       | 能看到参数         | **参数丢了，看不到** |
| Crash Dump | 参数在栈上         | 寄存器被覆盖，参数无 |

---

为什么 Release 优化掉？

  编译器认为：参数驻留的 mov 指令**浪费 CPU 周期**正常执行,不需要这些副本,栈空间也浪费,所以最终导致Release 版 crash 时，调试器只能看到当前寄存器值，看不到谁传的参数。

参数驻留 = Debug 版把寄存器参数复制到栈上"备份"，方便调试器查看。Release 版为了性能优化掉，所以 64 位 Release 程序崩溃时经常看不到函数参数。

## k命令函数的输出详解

我们前面学了32位和64位的栈帧输出和调用约定 现在解读一下k命令

  我们debug了32位release版的demo，在CdeclFunc下打入断点，输入命令kvn

![image-20260725035626709](/img/windbg笔记1/image-20260725035626709.png)

ChildEBP也就是该函数的EBP。Retaddr也就是返回地址 Arg to child就是函数的参数，默认只列出三个参数，多的列不出来

```
0000000a 00000014 0000001e
```

 这边十六进制换算成十进制就是：

0x0a = 0×16 + 10 = 10
0x14 = 1×16 + 4 = 20
0x1e = 1×16 + 14 = 30

  这里也对的上了。

![image-20260725042021169](/img/windbg笔记1/image-20260725042021169.png)

我们在输出dd ebp L6

![image-20260725042244369](/img/windbg笔记1/image-20260725042244369.png)

那么其实很明朗了

| 栈地址                  | 栈内容        | k 命令字段    | 说明                          |
| ----------------------- | ------------- | ------------- | ----------------------------- |
| `[ebp]` (00e3fa6c)      | 00e3fa9c      | ChildEBP      | 下一帧（main）的 EBP ← EBP 链 |
| `[ebp+4]` (00e3fa70)    | 00fe11f0      | RetAddr       | 返回地址                      |
| `[ebp+8]` (00e3fa74)    | 0000000a (10) | Args to Child | 参数 1                        |
| `[ebp+0xC]` (00e3fa78)  | 00000014 (20) | Args to Child | 参数 2                        |
| `[ebp+0x10]` (00e3fa7c) | 0000001e (30) | Args to Child | 参数 3                        |


 ChildEBP、RetAddr、Args to Child 三个字段都来自堆栈。

- ChildEBP 来自 [ebp] （被保存的旧 EBP）
- RetAddr 来自 [ebp+4] （call 指令隐式压入的返回地址）
- Args to Child 来自 [ebp+8] 、 [ebp+0xC] 、 [ebp+0x10] （调用前 push 压入的参数）
k 命令本质上就是 自动顺着 EBP 链爬整个调用栈，并把每个栈帧的固定字段格式化输出 。你完全可以用 dd 手动重建 k 的输出——这就是为什么 32 位下 k 命令如此可靠。

为什么 64 位下 k 不可靠

- 无 EBP ：ChildEBP 列无法填（没有 [ebp] 可读）
- 参数在寄存器 ：Args to Child 列无法从栈上读出
- 只有 RetAddr 相对可靠 （call 指令仍会压返回地址）
这就是笔记里"64 位全优化时只有 RetAddr 可信"的原因。



## WinDbg 地址范围与 dps 命令

  WinDbg 命令的"地址范围"参数语法 ，以及如何用 dps 命令导出整个线程栈。之前实验里已经在用这个语法了—— dd ebp L6 里的 L6 就是地址范围。

地址范围有两种写法：

方式 1：起始地址 + L对象数量

```
dd ebp L6        ; 你之前的实验
u  . L18         ; 反汇编 18 条
dps esp L3072    ; dump 3072 个指针
```

 方式 2：起始地址 + 结束地址

```
dps 008fd000 00900000   ; dump 从 008fd000 到 00900000 的内存
dd ebp ebp+0x18         ; 等价于 dd ebp L6
```

### 实操

  我们还是debug那个32位release的程序，在CdeclFunc下打入断点，打下断点输入命令：!teb

![image-20260725061652268](/img/windbg笔记1/image-20260725061652268.png)

| 字段                                   | 用途               | 调试场景                            |
| -------------------------------------- | ------------------ | ----------------------------------- |
| `StackBase`(栈底) / `StackLimit`(栈顶) | 定义当前线程栈范围 | 检查栈溢出、查看栈内存              |
| `ClientId`                             | 唯一标识进程和线程 | 区分线程、匹配任务管理器            |
| `PEB Address`                          | 指向进程全局信息   | `!peb` 查看模块列表、堆信息等       |
| `LastErrorValue`                       | Win32 API 错误码   | `!gle` 查看最后错误                 |
| `LastStatusValue`                      | 内核 NTSTATUS 码   | 分析系统调用失败原因                |
| `ExceptionList`                        | SEH 链起点         | 查看异常处理程序                    |
| `Self`                                 | TEB 自引用         | 通过 FS:\[0] / GS:\[0] 快速定位 TEB |

- StackBase = 00b00000（栈底，最高地址）
- StackLimit = 00afd000（栈顶，最低地址）
- 这里**ClientId 是 PID.TID**格式，所以00007268 . 0000818c
